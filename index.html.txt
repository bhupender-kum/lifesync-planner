            const now = new Date();

        /* -------------------------------------------------------------
           HISTORICAL ARCHIVE & DAILY RESET ENGINE
           ------------------------------------------------------------- */
        function checkAndProcessDailyArchive() {
            if (!appData) return;
            const todayKey = getTodayDateKey();
            const lastActiveDate = appData.lastActiveDate || todayKey;

            if (!appData.history) appData.history = {};

            // Archive status for last active day if it's a new day
            if (lastActiveDate !== todayKey) {
                const prevDate = lastActiveDate;
                const habits = appData.habits || [];
                const habitDoneCount = habits.filter(h => h.completedToday).length;
                const habitPct = habits.length > 0 ? Math.round((habitDoneCount / habits.length) * 100) : 0;

                const sched = appData.schedule || {};
                const schedTotal = Object.keys(sched).length;
                const schedDone = Object.values(sched).filter(s => s.completed).length;

                // Archive day snapshot into persistent history store
                appData.history[prevDate] = {
                    habitPct: habitPct,
                    habitsCompleted: habitDoneCount,
                    totalHabits: habits.length,
                    schedDone: schedDone,
                    schedTotal: schedTotal,
                    expensesTotal: (appData.expenses || [])
                        .filter(e => e.date === prevDate)
                        .reduce((sum, e) => sum + Number(e.amount), 0),
                    antiGoalsDefended: (appData.antiGoals || []).filter(ag => ag.avoided).length
                };

                // Reset daily status for today while retaining total streaks
                (appData.habits || []).forEach(h => {
                    h.completedToday = false;
                });
                (appData.toDos || []).forEach(t => {
                    t.completed = false;
                });
                (appData.antiGoals || []).forEach(ag => {
                    ag.avoided = false;
                });

                // Move active daily timetable into archive and clear daily active slots for the new day
                if (!appData.scheduleArchive) appData.scheduleArchive = {};
                appData.scheduleArchive[prevDate] = appData.schedule || {};
                appData.schedule = {};

                appData.lastActiveDate = todayKey;
                saveAppData();
            } else {
                appData.lastActiveDate = todayKey;
            }
        }

        async function generateTokenId(str) {
            try {
                if (window.crypto && window.crypto.subtle) {
                    const encoder = new TextEncoder();
                    const data = encoder.encode(str);
                    const hashBuffer = await window.crypto.subtle.digest('SHA-256', data);
                    const hashArray = Array.from(new Uint8Array(hashBuffer));
                    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('').substring(0, 24);
                }
            } catch (e) {}
            let hash = 0;
            for (let i = 0; i < str.length; i++) {
                hash = ((hash << 5) - hash) + str.charCodeAt(i);
                hash |= 0;
            }
            return "usr_" + Math.abs(hash).toString(16) + "x" + str.length;
        }

            const usernameInput = document.getElementById("authUsername");
            const passwordInput = document.getElementById("authPassword");

            if (!usernameInput || !passwordInput) return;

            const username = usernameInput.value.trim().toLowerCase();
            const password = passwordInput.value;

            if (!username || !password) return;

            let directory = {};
            try {
                directory = JSON.parse(localStorage.getItem("LIFESYNC_USERS_DIR") || "{}");
            } catch (err) {
                directory = {};
            }

            const token = await generateTokenId(username + ":" + password);

            if (mode === 'register') {
                if (directory[username]) {
                    showToast("Account already exists with this identifier. Switched to Sign In.", "alert-circle");
                    switchAuthTab('login');
                    passwordInput.value = "";
                    passwordInput.focus();
                    return;
                }
                const fullName = (document.getElementById("authFullName")?.value || "").trim() || username;
                const initBudget = parseFloat(document.getElementById("authInitBudget")?.value || "30000") || 30000;

                directory[username] = {
                    fullName: fullName,
                    tokenId: token,
                    created: new Date().toISOString()
                };
                localStorage.setItem("LIFESYNC_USERS_DIR", JSON.stringify(directory));

                appData = createFreshUserState(username, fullName, initBudget);
                appData.tokenId = token;
                localStorage.setItem(getUserStorageKey(token), JSON.stringify(appData));

                // Direct auto-login after successful registration
                completeLogin(username, fullName, token);
                showToast("Welcome to LifeSync OS!", "sparkles");
                triggerConfetti();
            } else {
                if (!directory[username]) {
                    showToast("No account found with this identifier. Please register first.", "alert-circle");
                    switchAuthTab('register');
                    return;
                }
                if (directory[username].tokenId !== token) {
                    showToast("Incorrect security passphrase!", "alert-circle");
                    passwordInput.value = "";
                    passwordInput.focus();
                    return;
                }
                const storageKey = getUserStorageKey(token);
                const storedData = localStorage.getItem(storageKey);
                if (storedData) {
                    try {
                        appData = JSON.parse(storedData);
                    } catch (err) {
                        appData = createFreshUserState(username, directory[username].fullName);
                    }
                } else {
                    appData = createFreshUserState(username, directory[username].fullName);
                }
                appData.tokenId = token;
                completeLogin(username, directory[username].fullName, token);
                showToast("Signed in successfully", "shield-check");
            }
        }

        function quickAddZoneTask(zone) {
            let inputId = "quickAddMorningInput";
            let defaultSlot = "08:00";
            if (zone === "afternoon") {
                inputId = "quickAddAfternoonInput";
                defaultSlot = "14:00";
            } else if (zone === "evening") {
                inputId = "quickAddEveningInput";
                defaultSlot = "20:00";
            }

            const inputEl = document.getElementById(inputId);
            if (!inputEl) return;
            const text = inputEl.value.trim();
            if (!text) return;

            const sched = appData.schedule || {};
            const rangeStart = zone === "morning" ? 6 : (zone === "afternoon" ? 12 : 18);
            const rangeEnd = zone === "morning" ? 12 : (zone === "afternoon" ? 18 : 23);

            let chosenHour = defaultSlot;
            for (let h = rangeStart; h < rangeEnd; h++) {
                const candidate = String(h).padStart(2, '0') + ":00";
                if (!sched[candidate]) {
                    chosenHour = candidate;
                    break;
                }
            }

            if (!appData.schedule) appData.schedule = {};
            appData.schedule[chosenHour] = {
                title: text,
                tag: zone === "morning" ? "Health" : (zone === "afternoon" ? "Work" : "Personal"),
                completed: false
            };

            inputEl.value = "";
            saveAppData();
            playTone("check");
            pokeMascot();
            showToast(`Added to ${zone} at ${chosenHour}!`, "clock");
        }

        function playTone(type = "check") {
            if (!soundEnabled) return;
            try {
                if (!audioContext) {
                    audioContext = new (window.AudioContext || window.webkitAudioContext)();
                }
                if (audioContext.state === "suspended") {
                    audioContext.resume();
                }

                const osc = audioContext.createOscillator();
                const gain = audioContext.createGain();
                osc.connect(gain);
                gain.connect(audioContext.destination);

                const now = audioContext.currentTime;

                if (type === "check") {
                    osc.type = "sine";
                    osc.frequency.setValueAtTime(587.33, now);
                    osc.frequency.exponentialRampToValueAtTime(880, now + 0.12);
                    gain.gain.setValueAtTime(0.12, now);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.15);
                    osc.start(now);
                    osc.stop(now + 0.15);
                } else if (type === "pop") {
                    osc.type = "triangle";
                    osc.frequency.setValueAtTime(329.63, now);
                    osc.frequency.exponentialRampToValueAtTime(659.25, now + 0.08);
                    gain.gain.setValueAtTime(0.15, now);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.09);
                    osc.start(now);
                    osc.stop(now + 0.09);
                } else if (type === "delete") {
                    osc.type = "sawtooth";
                    osc.frequency.setValueAtTime(220, now);
                    osc.frequency.exponentialRampToValueAtTime(110, now + 0.12);
                    gain.gain.setValueAtTime(0.08, now);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.13);
                    osc.start(now);
                    osc.stop(now + 0.13);
                }
            } catch (err) {}
        }

        function toggleSidebar(forceState) {
            const sidebar = document.getElementById("sidebar");
            const backdrop = document.getElementById("sidebarBackdrop");
            if (!sidebar || !backdrop) return;
            const isOpen = !sidebar.classList.contains("-translate-x-full");
            const shouldOpen = forceState !== undefined ? forceState : !isOpen;

            if (shouldOpen) {
                sidebar.classList.remove("-translate-x-full");
                backdrop.classList.remove("hidden");
            } else {
                sidebar.classList.add("-translate-x-full");
                backdrop.classList.add("hidden");

        }

        function updateGamificationUI() {
            const habits = appData.habits || [];
            const completedHabits = habits.filter(h => h.completedToday).length;
            const habitPct = habits.length > 0 ? Math.round((completedHabits / habits.length) * 100) : 0;

            const scheduledHours = Object.keys(appData.schedule || {}).length;
            const doneHours = Object.values(appData.schedule || {}).filter(ev => ev.completed).length;

            const toDos = appData.toDos || [];
            const doneToDos = toDos.filter(t => t.completed).length;

            const totalScore = Math.min(100, Math.round((habitPct * 0.45) + ((doneHours / Math.max(1, scheduledHours)) * 35) + ((doneToDos / Math.max(1, toDos.length)) * 20)));

            const streakCount = appData.streakDays || 0;
            const streakDisplay = document.getElementById("streakCount");
            const mobileStreak = document.getElementById("mobileStreakCount");
            if (streakDisplay) streakDisplay.textContent = `${streakCount} Days Streak`;
            if (mobileStreak) mobileStreak.textContent = `${streakCount}d`;

            const circle = document.getElementById("dashCircleProgress");
            const pctText = document.getElementById("dashCirclePercent");
            if (circle) {
                const circumference = 2 * Math.PI * 19;
                const offset = circumference - (totalScore / 100) * circumference;
                circle.style.strokeDashoffset = offset;
            }
            if (pctText) pctText.textContent = `${totalScore}%`;

            const level = Math.max(1, Math.floor((appData.xp || 0) / 100) + 1);
            const levelEl = document.getElementById("dashLevelNumber");
            if (levelEl) levelEl.textContent = `LVL ${level}`;

            const sideBadge = document.getElementById("sidebarLevelBadge");
            if (sideBadge) {
                const ranks = ["Novice", "Initiate", "Disciple", "Warrior", "Master", "Grandmaster"];
                const rankName = ranks[Math.min(ranks.length - 1, level - 1)];
                sideBadge.innerHTML = `<i data-lucide="award" class="w-3 h-3"></i> ${rankName} • ${appData.xp || 0} XP`;
            }
        }

                    appData.history[todayKey].habitPct = currentPct;

                    appData.history[todayKey] = currentPct;

        function copyRawStorage(key) {
            const val = localStorage.getItem(key);
            if (!val) return;
            const input = document.createElement("textarea");
            input.value = val;
            document.body.appendChild(input);
            input.select();
            document.execCommand("copy");
            document.body.removeChild(input);
            showToast(`Copied ${key} to clipboard`, "check");
            playTone("check");
        }

        function checkNotificationPermissionState() {
            const bellText = document.getElementById("nativeBellText");
            const badge = document.getElementById("pushStatusBadge");
            const bellIcon = document.getElementById("nativeBellIcon");

            if (!("Notification" in window)) {
                if (bellText) bellText.textContent = "Unsupported";
                if (badge) badge.textContent = "N/A";
                return;
            }

            if (Notification.permission === "granted") {
                if (bellText) bellText.textContent = "Alerts ON";
                if (badge) {
                    badge.textContent = "Active";
                    badge.className = "text-[10px] font-mono font-bold px-2 py-0.5 rounded-full bg-emerald-500/20 text-emerald-500 border border-emerald-500/30";
                }
                if (bellIcon) {
                    bellIcon.classList.remove("text-amber-500");
                    bellIcon.classList.add("text-emerald-500");
                }
            } else if (Notification.permission === "denied") {
                if (bellText) bellText.textContent = "Blocked";
                if (badge) {
                    badge.textContent = "Blocked in Browser";
                    badge.className = "text-[10px] font-mono font-bold px-2 py-0.5 rounded-full bg-red-500/20 text-red-500";
                }
            } else {
                if (bellText) bellText.textContent = "Enable Alerts";
                if (badge) {
                    badge.textContent = "Not Enabled";
                    badge.className = "text-[10px] font-mono font-bold px-2 py-0.5 rounded-full bg-slate-200 dark:bg-charcoal-800 text-slate-500";
                }
            }
        }

        async function requestPhonePushPermission() {
            if (!("Notification" in window)) {
                showToast("Your browser does not support web notifications.", "alert-triangle");

            try {
                const permission = await Notification.requestPermission();
                checkNotificationPermissionState();
                if (permission === "granted") {

                    playTone("check");
                    triggerNativeNotification("LifeSync Alert Engine Online", "You will now receive focus reminders directly on your phone screen.");

        function createGoogleCalendarUrl(title, details, startTimeStr, endTimeStr, dateObj = new Date()) {
            const y = dateObj.getFullYear();
            const m = String(dateObj.getMonth() + 1).padStart(2, '0');
            const d = String(dateObj.getDate()).padStart(2, '0');

            const startHour = startTimeStr ? startTimeStr.split(':')[0].padStart(2, '0') : "09";
            const startMin = startTimeStr && startTimeStr.split(':')[1] ? startTimeStr.split(':')[1] : "00";
            
            const endHour = endTimeStr ? endTimeStr.split(':')[0].padStart(2, '0') : String((parseInt(startHour, 10) + 1) % 24).padStart(2, '0');
            const endMin = endTimeStr && endTimeStr.split(':')[1] ? endTimeStr.split(':')[1] : "00";

            const dtStart = `${y}${m}${d}T${startHour}${startMin}00`;
            const dtEnd = `${y}${m}${d}T${endHour}${endMin}00`;

            const encodedTitle = encodeURIComponent(`[LifeSync] ${title}`);
            const encodedDetails = encodeURIComponent(`${details}\n\nManaged in LifeSync Daily Planner.`);

            return `https://calendar.google.com/calendar/render?action=TEMPLATE&text=${encodedTitle}&details=${encodedDetails}&dates=${dtStart}/${dtEnd}&ctz=Asia/Kolkata`;
        }

        function downloadCalendarIcs() {
            const sched = appData.schedule || {};
            const hours = Object.keys(sched);
            if (hours.length === 0) {
                showToast("Schedule at least one time block first!", "alert-circle");
                return;
            }

            const now = new Date();
            const y = now.getFullYear();
            const m = String(now.getMonth() + 1).padStart(2, '0');
            const d = String(now.getDate()).padStart(2, '0');

            let icsContent = "BEGIN:VCALENDAR\nVERSION:2.0\nPRODID:-//LifeSync OS//Daily Schedule//EN\nCALSCALE:GREGORIAN\n";

            hours.forEach(hourStr => {
                const item = sched[hourStr];
                const startHour = parseInt(hourStr.split(':')[0], 10);
                const endHour = (startHour + 1) % 24;

                const dtStart = `${y}${m}${d}T${String(startHour).padStart(2, '0')}0000`;
                const dtEnd = `${y}${m}${d}T${String(endHour).padStart(2, '0')}0000`;

                icsContent += `BEGIN:VEVENT\n`;
                icsContent += `SUMMARY:[LifeSync] ${item.title} (${item.tag})\n`;
                icsContent += `DESCRIPTION:LifeSync scheduled focus block for ${item.tag}.\n`;
                icsContent += `DTSTART:${dtStart}\n`;
                icsContent += `DTEND:${dtEnd}\n`;
                icsContent += `STATUS:${item.completed ? 'COMPLETED' : 'CONFIRMED'}\n`;
                icsContent += `BEGIN:VALARM\nTRIGGER:-PT10M\nACTION:DISPLAY\nDESCRIPTION:Upcoming Focus Block: ${item.title}\nEND:VALARM\n`;
                icsContent += `END:VEVENT\n`;
            });

            icsContent += "END:VCALENDAR";

            const blob = new Blob([icsContent], { type: "text/calendar;charset=utf-8" });
            const link = document.createElement("a");
            link.href = URL.createObjectURL(blob);
            link.download = `LifeSync_Schedule_${y}-${m}-${d}.ics`;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);

            closePhoneSyncModal();
            showToast("Calendar (.ics) downloaded!", "calendar");
            playTone("check");
        }

                modal.classList.remove("hidden");
                lucide.createIcons();

                }
            });
        }

        function exportDataJson() {
            try {
                const payload = {
                    ...appData,
                    exportedAt: new Date().toISOString(),
                    currency: "INR",
                    version: "4.0-lifesync-os"
                };
                const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(payload, null, 2));
                const downloadAnchor = document.createElement('a');
                downloadAnchor.setAttribute("href", dataStr);
                const dateStr = new Date().toISOString().split('T')[0];
                downloadAnchor.setAttribute("download", `LifeSync_${currentUser ? currentUser.username : 'data'}_${dateStr}.json`);
                document.body.appendChild(downloadAnchor);
                downloadAnchor.click();
                downloadAnchor.remove();
                showToast("Backup downloaded successfully", "download");
                playTone("check");
            } catch (err) {
                showToast("Failed to generate backup file", "alert-circle");
            }
        }

        function importDataJson(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(event) {
                try {
                    const parsed = JSON.parse(event.target.result);
                    if (parsed && typeof parsed === 'object') {
                        appData = { ...appData, ...parsed };
                        saveAppData();
                        showToast("Backup restored successfully!", "upload");
                        playTone("check");
                        triggerConfetti();
                    } else {
                        showToast("Invalid backup file format", "alert-circle");
                    }
                } catch (err) {
                    showToast("Malformed backup file", "alert-circle");
                }
            };
            reader.readAsText(file);
            e.target.value = "";
        }

        function escapeHtml(text) {
            if (!text) return "";
            return String(text)
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }

            if (savedPhone && phoneInput) {
                phoneInput.value = savedPhone;
            }

            // Persistence check: Active token lookup from localStorage
            const activeToken = localStorage.getItem("LIFESYNC_ACTIVE_TOKEN");
            const activeUser = localStorage.getItem("LIFESYNC_ACTIVE_USER");

            let autoLoggedIn = false;

            if (activeToken && activeUser) {
                let directory = {};
                try {
                    directory = JSON.parse(localStorage.getItem("LIFESYNC_USERS_DIR") || "{}");
                } catch (e) { directory = {}; }

                if (directory[activeUser] && directory[activeUser].tokenId === activeToken) {
                    const storageKey = getUserStorageKey(activeToken);
                    const stored = localStorage.getItem(storageKey);
                    if (stored) {
                        try {
                            appData = JSON.parse(stored);
                        } catch (e) {
                            appData = createFreshUserState(activeUser, directory[activeUser].fullName);
                        }
                    } else {
                        appData = createFreshUserState(activeUser, directory[activeUser].fullName);
                    }
                    appData.tokenId = activeToken;
                    completeLogin(activeUser, directory[activeUser].fullName || activeUser, activeToken);
                    autoLoggedIn = true;
                }
            }

            if (!autoLoggedIn) {
                const authOverlay = document.getElementById("authOverlay");
                if (authOverlay) authOverlay.classList.remove("hidden");

                let directory = {};
                try {
                    directory = JSON.parse(localStorage.getItem("LIFESYNC_USERS_DIR") || "{}");
                } catch (e) { directory = {}; }

                if (activeUser && directory[activeUser]) {
                    const usernameInput = document.getElementById("authUsername");
                    if (usernameInput) usernameInput.value = activeUser;
                }

                if (Object.keys(directory).length > 0) {
                    switchAuthTab('login');
                } else {
                    switchAuthTab('register');
                }
            } else {
                checkDailyPlanningAlerts();
            }

            initTrafficAnalytics();

            const hash = window.location.hash.replace('#', '');
            if (hash && ALL_VIEWS.includes(hash)) {
                navigate(hash);
            }
            lucide.createIcons();
        };
